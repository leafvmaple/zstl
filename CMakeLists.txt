cmake_minimum_required(VERSION 3.16)
project(zstl LANGUAGES CXX)

add_library(zstl INTERFACE)
add_library(zstl::zstl ALIAS zstl)

target_include_directories(zstl INTERFACE
  "${CMAKE_CURRENT_SOURCE_DIR}/include"
)

target_compile_features(zstl INTERFACE cxx_std_17)

option(ZSTL_BUILD_TESTS "Build zstl contract tests" OFF)
if(ZSTL_BUILD_TESTS)
  enable_testing()
  add_executable(zstl_memory_test tests/memory_test.cpp)
  target_link_libraries(zstl_memory_test PRIVATE zstl::zstl)
  add_test(NAME zstl_memory COMMAND zstl_memory_test)
  add_executable(zstl_container_test tests/container_test.cpp)
  target_link_libraries(zstl_container_test PRIVATE zstl::zstl)
  target_compile_features(zstl_container_test PRIVATE cxx_std_20)
  add_test(NAME zstl_containers COMMAND zstl_container_test)
  add_executable(zstl_cstring_test tests/cstring_test.cpp)
  target_link_libraries(zstl_cstring_test PRIVATE zstl::zstl)
  target_compile_definitions(zstl_cstring_test PRIVATE ZSTL_FREESTANDING)
  add_test(NAME zstl_cstring COMMAND zstl_cstring_test)
  add_library(zstl_container_freestanding_test OBJECT tests/container_freestanding_test.cpp)
  target_link_libraries(zstl_container_freestanding_test PRIVATE zstl::zstl)
  target_compile_features(zstl_container_freestanding_test PRIVATE cxx_std_20)
  target_compile_definitions(zstl_container_freestanding_test PRIVATE ZSTL_FREESTANDING)
  target_compile_options(zstl_container_freestanding_test PRIVATE -ffreestanding -nostdinc -nostdinc++ -fno-exceptions -fno-rtti)
  add_library(zstl_freestanding_test OBJECT tests/freestanding_test.cpp)
  target_link_libraries(zstl_freestanding_test PRIVATE zstl::zstl)
  target_compile_definitions(zstl_freestanding_test PRIVATE ZSTL_FREESTANDING)
  target_compile_options(zstl_freestanding_test PRIVATE -ffreestanding -nostdinc -nostdinc++ -fno-exceptions -fno-rtti)
endif()
